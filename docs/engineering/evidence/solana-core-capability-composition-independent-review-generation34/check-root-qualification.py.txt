def git(*args):
    return subprocess.check_output(['git', *args], cwd=REVIEW)


def sha(value):
    return hashlib.sha256(value).hexdigest()


def preserve(name, raw):
    compressed = gzip.compress(raw, mtime=0)
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists():
        assert target.read_bytes() == compressed, 'Never overwrite differing retained evidence'
    else:
        target.write_bytes(compressed)
    assert gzip.decompress(compressed) == raw and int.from_bytes(compressed[4:8], 'little') == 0
    return {'path': EVIDENCE + '/' + name, 'rawBytes': len(raw), 'rawSha256': sha(raw),
            'bytes': len(compressed), 'sha256': sha(compressed), 'gzipMtime': 0}


def entries(ref):
    result = {}
    for row in git('ls-tree', '-r', '-z', ref).split(b'\0'):
        if row:
            meta, path = row.split(b'\t', 1)
            result[path.decode()] = meta.decode()
    return result


def nondoc(value):
    return {path: meta for path, meta in value.items() if not path.startswith('docs/')}
