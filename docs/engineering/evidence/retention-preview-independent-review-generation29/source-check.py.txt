from pathlib import Path
import hashlib
import json
import subprocess
import sys

ROOT = Path('/workspace/scratch/dcf03ef9bee8/p2p-retention-preview-review-g29')
DEPENDENCIES = Path('/workspace/scratch/dcf03ef9bee8/p2p-continuation-g26')
BASE = '76d107e4e0ca1a2533636a6057d9d603861668e1'
PUBLISHED_BASE = 'd9ddcf53d82100f05de7346143a0ad00d43c2c1a'
BACKEND = '5ad059fa8b77cfb8beffcf2f1889e5f19d727ea4'

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT).decode().strip()

def entries(commit):
    data = subprocess.check_output(['git', 'ls-tree', '-rz', '--full-tree', commit], cwd=ROOT)
    return dict(item.decode().split('\t', 1)[::-1] for item in data.split(b'\0') if item)

def sha(data):
    return hashlib.sha256(data).hexdigest()

expected_source, allow_path = sys.argv[1:]
allowed = json.loads(Path(allow_path).read_text())
actual = git('rev-parse', 'HEAD')
before, after = entries(BASE), entries(actual)
changed = sorted(path for path in before.keys() | after.keys() if before.get(path) != after.get(path))
status = git('status', '--porcelain=v1', '--untracked-files=all')
checks = {
    'exact_frozen_source': actual == expected_source,
    'clean_source_checkout': status == '',
    'full_changed_path_allowlist': changed == sorted(allowed),
    'no_deleted_baseline_entry': not (before.keys() - after.keys()),
    'every_unlisted_baseline_entry_preserved': all(after.get(p) == entry for p, entry in before.items() if p not in allowed),
    'base_tree_is_assigned_tree': git('rev-parse', BASE + '^{tree}') == '437323a382952b9623a875c818f8af005a5d4563',
    'backend_tree_is_assigned_tree': git('rev-parse', BACKEND + '^{tree}') == '05ff7a561aea207d6d54dad08d08fadd5cd8cf45',
    'physical_dependency_root': (ROOT / 'node_modules').is_dir() and not (ROOT / 'node_modules').is_symlink(),
    'dependency_lock_matches_source_copy': (ROOT / 'package-lock.json').read_bytes() == (DEPENDENCIES / 'package-lock.json').read_bytes(),
}
try:
    published_tree = git('rev-parse', PUBLISHED_BASE + '^{tree}')
except subprocess.CalledProcessError:
    published_tree = None
if published_tree:
    checks['published_base_tree_matches_local_base'] = published_tree == git('rev-parse', BASE + '^{tree}')
files = []
for path in changed + ['packages/storage/src/retention-service.ts', 'packages/storage/test/retention-service.test.mjs',
    'apps/web/test/unlock-form-admission.test.mjs', 'apps/web/test/vault-form-admission.test.mjs',
    'scripts/test-update-runtime-browser.mjs', 'package.json', 'package-lock.json',
    '00_PLANNING_AND_THREAT_MODELING/feature_0.8_data_governance/module_0.8.1_data_lifecycle/spec-data-retention-matrix.md']:
    data = (ROOT / path).read_bytes()
    files.append({'path': path, 'git_entry': after[path], 'bytes': len(data), 'sha256': sha(data)})
dependencies = []
for name in ['react', 'react-dom', 'vite', 'typescript', '@noble/ciphers', '@noble/curves', '@noble/hashes']:
    local = ROOT / 'node_modules' / name / 'package.json'
    origin = DEPENDENCIES / 'node_modules' / name / 'package.json'
    data = local.read_bytes()
    dependencies.append({'name': name, 'version': json.loads(data)['version'], 'local_path': str(local),
        'real_path': str(local.resolve()), 'sha256': sha(data), 'origin_path': str(origin),
        'same_inode_as_origin': local.stat().st_ino == origin.stat().st_ino and local.stat().st_dev == origin.stat().st_dev})
checks['dependencies_are_recorded_immutable_hardlinks'] = all(d['same_inode_as_origin'] for d in dependencies)
report = {
    'source': actual, 'tree': git('rev-parse', 'HEAD^{tree}'), 'base': BASE, 'base_tree': git('rev-parse', BASE + '^{tree}'),
    'published_base': PUBLISHED_BASE, 'published_base_tree': published_tree,
    'baseline_entries': len(before), 'source_entries': len(after), 'changed_paths': changed,
    'unchanged_baseline_entries': sum(after.get(path) == value for path, value in before.items()),
    'checks': checks, 'working_tree_status': status, 'files': files, 'dependencies': dependencies,
    'dependency_preparation': 'cp -al from the frozen-base checkout; local directory paths share immutable inodes; no installs or package mutations',
    'node': subprocess.check_output(['node', '--version'], text=True).strip(),
}
print(json.dumps(report, indent=2))
assert all(checks.values()), 'source identity or scope check failed'
