import test from 'node:test';
import assert from 'node:assert/strict';
import { pathToFileURL } from 'node:url';

const moduleAt = relative => import(pathToFileURL(`${process.cwd()}/${relative}`).href);
const { connected, setupClient } = await moduleAt('packages/chain-solana/test/lifecycle-support.mjs');
const { error, key, ok, PROGRAM_DATA } = await moduleAt('packages/chain-solana/test/support.mjs');
const deferred = () => {
  let resolve;
  const promise = new Promise(done => { resolve = done; });
  return { promise, resolve };
};
const sends = env => env.cluster.calls.filter(call => call.method === 'sendTransaction').length;
const records = env => env.journalFake.records(env.binding);

test('independent: queued SPL funding captures its source before a cancelled physical wallet slot settles', async t => {
  const entered = deferred(), held = deferred();
  const env = setupClient('spl', { clusterOptions: { absent: true }, wallet: {
    output: async wire => { entered.resolve(); await held.promise; return [{ signedTransaction: wire }]; },
  } });
  t.after(() => { held.resolve(); env.client.dispose(); });
  const session = await connected(env, 'seller');
  const first = ok(await env.client.prepareFunding(env.trade, session));
  const confirming = env.client.confirm(first, session);
  await entered.promise;
  const cancelling = env.client.cancel(first);
  const captured = key('independent-g32-captured-source');
  const options = Object.assign(Object.create(null), { sellerTokenAccount: captured });
  let nextSettled = false;
  const next = env.client.prepareFunding(env.trade, session, options).then(result => {
    nextSettled = true;
    return result;
  });
  options.sellerTokenAccount = key('independent-g32-mutated-source');
  options.rebuildOf = 'not-an-intent';
  await new Promise(resolve => setImmediate(resolve));
  assert.equal(nextSettled, false);
  assert.equal(env.walletFake.calls.sign.length, 1);
  assert.equal(sends(env), 0);
  held.resolve();
  error(await confirming, 'CANCELLED');
  ok(await cancelling);
  const review = ok(await next);
  assert.equal(review.action, 'fund');
  assert.equal(review.shape.sellerTokenAccount, captured);
  assert.notEqual(review.intentId, first.intentId);
  assert.deepEqual(records(env).map(record => record.state), ['cancelled', 'prepared']);
  assert.equal(env.walletFake.calls.sign.length, 1);
  assert.equal(sends(env), 0);
  ok(await env.client.cancel(review));
});

test('independent: synchronous funding-option observation cannot continue after disposing the client', async () => {
  const env = setupClient('native', { clusterOptions: { absent: true } });
  const session = await connected(env, 'seller');
  let traps = 0;
  const options = new Proxy({}, {
    ownKeys() { traps++; env.client.dispose(); return []; },
  });
  error(await env.client.prepareFunding(env.trade, session, options), 'CANCELLED');
  assert.ok(traps > 0);
  assert.equal(env.cluster.calls.length, 0);
  assert.deepEqual(env.journalFake.writes, []);
  assert.equal(env.walletFake.calls.sign.length, 0);
});

test('independent: profile replacement immediately after funding-intent persistence publishes no capability', async t => {
  let env;
  env = setupClient('native', { clusterOptions: { absent: true }, journal: { hooks: {
    after(record) { if (record.state === 'prepared') env.journalFake.setProfile('b'.repeat(32)); },
  } } });
  t.after(() => env.client.dispose());
  const session = await connected(env, 'seller');
  const result = await env.client.prepareFunding(env.trade, session);
  error(result, 'LOCKED');
  assert.equal('value' in result, false);
  assert.equal(records(env).length, 1);
  assert.equal(records(env)[0].originProfileId, 'a'.repeat(32));
  assert.equal(records(env)[0].state, 'prepared');
  assert.equal(records(env)[0].signature, null);
  assert.equal(records(env)[0].sendMarkedAt, null);
  assert.equal(env.walletFake.calls.sign.length, 0);
  assert.equal(sends(env), 0);
});

test('independent: changed deployment bytes during a funding wallet prompt prevent a durable send marker', async t => {
  const entered = deferred(), held = deferred();
  const env = setupClient('native', { clusterOptions: { absent: true }, wallet: {
    output: async wire => { entered.resolve(); await held.promise; return [{ signedTransaction: wire }]; },
  } });
  t.after(() => { held.resolve(); env.client.dispose(); });
  const session = await connected(env, 'seller');
  const review = ok(await env.client.prepareFunding(env.trade, session));
  const confirming = env.client.confirm(review, session);
  await entered.promise;
  const old = env.cluster.world.accounts.get(PROGRAM_DATA);
  const bytes = Buffer.from(old.data[0], 'base64');
  bytes[45] ^= 1;
  env.cluster.world.accounts.set(PROGRAM_DATA, { ...old, data: [bytes.toString('base64'), 'base64'] });
  held.resolve();
  error(await confirming, 'INTEGRITY_FAILURE');
  assert.equal(env.walletFake.calls.sign.length, 1);
  assert.equal(sends(env), 0);
  assert.equal(records(env)[0].state, 'cancelled');
  assert.equal(records(env)[0].sendMarkedAt, null);
  assert.equal(records(env)[0].message, review.message);
  assert.equal(records(env)[0].blockhash, review.blockhash);
});
