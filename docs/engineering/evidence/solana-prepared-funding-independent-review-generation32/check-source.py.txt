import hashlib
import json
import subprocess
import sys
from pathlib import Path

REPO = Path('/workspace/scratch/dcf03ef9bee8/p2p-solana-prepared-funding-review-g32')
BASE = '8d566f0e46ee76ee35a957cab0a4358a3da4aeb3'
source = sys.argv[1]


def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)


def entries(ref):
    result = {}
    for row in git('ls-tree', '-r', '-z', ref).split(b'\0'):
        if row:
            identity, path = row.split(b'\t', 1)
            result[path.decode()] = identity.decode()
    return result


assert git('rev-parse', 'HEAD').decode().strip() == source
assert not git('status', '--porcelain').strip()
assert git('show', '-s', '--format=%P', source).decode().strip() == BASE
base, current = entries(BASE), entries(source)
changed = sorted(path for path in base.keys() | current.keys() if base.get(path) != current.get(path))
expected = sorted([
    'packages/chain-solana/src/compose.ts',
    'packages/chain-solana/test/prepared-funding.test.mjs',
    'packages/chain-solana/README.md',
])
assert changed == expected
assert not base.keys() - current.keys()
assert current['packages/chain-solana/test/prepared-actions.test.mjs'] == base['packages/chain-solana/test/prepared-actions.test.mjs']
protected_prefixes = ['apps/web/src/solana-localnet/', 'apps/web/src/runtime/', 'contracts/solana/',
                      'packages/chain-solana/src/']
for path, identity in base.items():
    if any(path.startswith(prefix) for prefix in protected_prefixes) and path not in expected:
        assert current[path] == identity
subprocess.run(['git', 'diff', '--check', BASE + '..' + source], cwd=REPO, check=True)
print(json.dumps({
    'base': BASE,
    'source': source,
    'tree': git('rev-parse', source + '^{tree}').decode().strip(),
    'changed': changed,
    'unchangedBaselineEntries': sum(current.get(path) == identity for path, identity in base.items()),
    'nonDocumentationEntries': sum(not path.startswith('docs/') for path in current),
    'entries': {path: {'identity': current[path], 'sha256': hashlib.sha256(git('show', source + ':' + path)).hexdigest()}
                for path in changed},
    'allOtherEntriesPreserved': True,
    'unchangedExitPreparedTests': True,
    'browserProtocolAndNativeProgramUnchanged': True,
}, indent=2))
