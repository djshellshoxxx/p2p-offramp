// All mounts in one world share one evaluation of the maintained leaf, including
// its module-level physical-write admission. Effects and memo dependencies are
// scheduled independently per mount. This is not a browser or DOM fixture.
function world(t, mode = 'held') {
  let current = null, clipboardReads = 0, permissionsReads = 0, shareReads = 0, networkCalls = 0;
  const attempts = [], mounts = new Set();
  const clipboard = {
    writeText(text) {
      assert.equal(this, clipboard);
      const gate = deferred(); attempts.push({ text, ...gate });
      if (mode === 'reject') return Promise.reject(new Error('PRIVATE browser detail'));
      if (mode === 'throw') throw new Error('PRIVATE browser detail');
      return gate.promise;
    },
  };
  const environment = { TextEncoder, TextDecoder,
    navigator: {
      get clipboard() {
        clipboardReads++;
        if (mode === 'getter-throws') throw new Error('PRIVATE getter detail');
        if (mode === 'missing') return undefined;
        if (mode === 'noncallable') return { writeText: {} };
        return clipboard;
      },
      get permissions() { permissionsReads++; throw new Error('permission probing is outside this operation'); },
      get share() { shareReads++; throw new Error('native sharing is outside this operation'); },
    },
    fetch() { networkCalls++; throw new Error('network is outside this operation'); },
    hooks: {
      useState(initial) { return current.state(initial); },
      useRef(initial) { return current.ref(initial); },
      useMemo(factory, dependencies) { return current.memo(factory, dependencies); },
      useEffect(callback, dependencies) { return current.effect(callback, dependencies); },
      useSyncExternalStore(subscribe, snapshot) {
        const owner = current;
        owner.effect(() => subscribe(() => { owner.dirty = true; }), [subscribe]);
        return snapshot();
      },
    },
  };
  runInNewContext(compiled, environment);
  const api = environment.independentShareReview, m = api.messageGroup('shell');
  function mount(initial = A) {
    let appLink = initial, stateAt = 0, refAt = 0, memoAt = 0, effectAt = 0, active = true, tree;
    const states = [], refs = [], memos = [], effects = [];
    const ui = {
      dirty: true, writesAfterUnmount: 0,
      state(initial) {
        const at = stateAt++;
        if (!(at in states)) states[at] = typeof initial === 'function' ? initial() : initial;
        return [states[at], value => {
          if (!active) { ui.writesAfterUnmount++; return; }
          const next = typeof value === 'function' ? value(states[at]) : value;
          if (!Object.is(next, states[at])) { states[at] = next; ui.dirty = true; }
        }];
      },
      ref(initial) { return refs[refAt++] ??= { current: initial }; },
      memo(factory, dependencies) {
        const at = memoAt++;
        if (!memos[at] || !same(memos[at].dependencies, dependencies)) memos[at] = { dependencies, value: factory() };
        return memos[at].value;
      },
      effect(callback, dependencies) {
        const at = effectAt++, prior = effects[at];
        if (!prior || !same(prior.dependencies, dependencies)) effects[at] = { callback, dependencies, stop: prior?.stop, pending: true };
      },
      view() {
        assert.equal(active, true, 'cannot render an unmounted fixture');
        for (let step = 0; step < 12; step++) {
          if (ui.dirty) {
            ui.dirty = false; stateAt = refAt = memoAt = effectAt = 0; current = ui;
            try { tree = api.GenericAppShare({ appLink }); } finally { current = null; }
          }
          for (const item of effects) if (item.pending) { item.pending = false; item.stop?.(); item.stop = item.callback(); }
          if (!ui.dirty) return tree;
        }
        assert.fail('independent leaf fixture failed to settle');
      },
      button() { return nodes(ui.view(), node => node.type === 'button')[0]; },
      input() { return nodes(ui.view(), node => node.type === 'input')[0]; },
      statuses() { return nodes(ui.view(), node => node.props.role === 'status').map(words); },
      click() { const button = ui.button(); assert.ok(button); assert.notEqual(button.props.disabled, true); button.props.onClick(); },
      replace(next) { appLink = next; ui.dirty = true; return ui.view(); },
      replayEffects() {
        for (const item of effects) item.stop?.();
        for (const item of effects) item.stop = item.callback();
      },
      unmount() { if (active) { for (const item of effects) item.stop?.(); active = false; mounts.delete(ui); } },
    };
    mounts.add(ui); t.after(ui.unmount); ui.view(); return ui;
  }
  return { mount, attempts, m, api,
    get clipboardReads() { return clipboardReads; },
    select(id) { api.localeStore.select(id); for (const ui of mounts) ui.dirty = true; },
    assertNoOtherSideEffects() { assert.equal(permissionsReads, 0); assert.equal(shareReads, 0); assert.equal(networkCalls, 0); },
  };
}

test('independent: real Settings exposes the supplied bytes and null fallback without startup clipboard access', () => {
  const before = Object.getOwnPropertyDescriptor(globalThis, 'navigator'); let reads = 0;
  Object.defineProperty(globalThis, 'navigator', { configurable: true, get() { reads++; throw new Error('startup clipboard access'); } });
  try {
    const html = renderToStaticMarkup(createElement(Settings, { appLink: A }));
    assert.match(html, /Share this app/); assert.ok(html.includes('value="' + A + '"'));
    assert.match(html, /readOnly=""/); assert.match(html, /Copy app address/);
    for (const props of [{}, { appLink: null }]) {
      const fallback = renderToStaticMarkup(createElement(Settings, props));
      assert.match(fallback, /A generic app address is unavailable/);
      assert.equal(fallback.includes('id="generic-app-url"'), false);
      assert.equal(fallback.includes('Copy app address'), false);
    }
    assert.equal(reads, 0);
  } finally { if (before) Object.defineProperty(globalThis, 'navigator', before); else delete globalThis.navigator; }
});

test('independent: held physical write blocks simultaneous and replacement mounts until settlement', async t => {
  const w = world(t), first = w.mount(), second = w.mount(B);
  const otherClick = second.button().props.onClick; first.click(); otherClick();
  assert.equal(w.attempts.length, 1); assert.equal(second.button().props.disabled, true);
  first.unmount(); const replacement = w.mount(B); replacement.button().props.onClick();
  assert.equal(w.attempts.length, 1); assert.equal(replacement.button().props.disabled, true);
  w.attempts[0].resolve(); await turn(); assert.equal(first.writesAfterUnmount, 0);
  assert.deepEqual(second.statuses(), ['']); assert.deepEqual(replacement.statuses(), ['']);
  assert.equal(replacement.button().props.disabled, false); replacement.click();
  assert.equal(w.attempts.length, 2); assert.equal(w.attempts[1].text, B);
  w.attempts[1].resolve(); await turn(); assert.deepEqual(replacement.statuses(), [w.m.shareAppCopied]);
  assert.deepEqual(second.statuses(), ['']); w.assertNoOtherSideEffects();
});
