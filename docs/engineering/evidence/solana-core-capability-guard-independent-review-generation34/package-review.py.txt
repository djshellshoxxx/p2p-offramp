import datetime
import gzip
import hashlib
import json
import subprocess
from pathlib import Path

ROOT = Path('/workspace/scratch/dcf03ef9bee8')
REPO = ROOT / 'p2p-solana-core-capability-guard-review-g34'
OUT = Path(__file__).resolve().parent
SOURCE = '2c21958684e8943a3058b0195fd310bfeba68866'
AUTHOR = '853c0b45607eb30f5e438483450c7ab763e6bcfb'
BASE = '448cdcc2a9d13109421f21eaa1107936da0ac5c8'
MAIN = '8e836baa877b676a863c877003e277b638fc5061'

TARGET = REPO / REL
CHECKS = ['targeted-tests', 'package-typecheck', 'source-and-whitespace', 'static-call-graph',
          'author-evidence-verification']


def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


assert git('rev-parse', 'HEAD').decode().strip() == SOURCE
assert not git('status', '--porcelain').strip()
assert not TARGET.exists()
TARGET.mkdir(parents=True)
artifacts = []


def copy(name, destination=None, compress=False):
    original = (OUT / name).read_bytes()
    value = gzip.compress(original, mtime=0) if compress else original
    filename = destination or name
    (TARGET / filename).write_bytes(value)
    artifacts.append({'path': (REL / filename).as_posix(), 'bytes': len(value), 'sha256': sha(value),
                      'rawBytes': len(original), 'rawSha256': sha(original),
                      'gzipMtime': 0 if compress else None})


checks = []
for name in CHECKS:
    record = json.loads((OUT / (name + '.json')).read_bytes())
    raw = (OUT / (name + '.log')).read_bytes()
    assert record['source'] == record['finalSource'] == SOURCE
    assert record['tree'] == 'e37f179bf8af392d884840fb083bc10a5ebff9f9'
    assert record['cleanTrackedWorktree'] and record['exitCode'] == 0
    assert len(raw) == record['bytes'] and sha(raw) == record['sha256']
    checks.append(record)
    copy(name + '.json')
    copy(name + '.log', name + '.log.gz', compress=True)

        'No second admitted production signing surface was found; static reference inventory is not a dynamic reachability proof.',
        'G33 historical data consistency remains separate from current funding eligibility and confirmation authority.',
        'Existing dependencies were reused as immutable physical hardlinks; no install, dependency edit or remote mutation occurred.',
