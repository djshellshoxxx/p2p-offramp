import hashlib
import json
import subprocess
import tempfile
from pathlib import Path

REPO = Path('/workspace/scratch/dcf03ef9bee8/p2p-solana-prepared-funding-review-g32')
BASE = '252dd1603f12eab69313ef25b538be1b040bc508'
SOURCE = 'e214f70f8bcbb9eb962fc147f2d318081daca2ce'
PATH = 'contracts/solana/localnet/client-run.mjs'


def git(*args):
    return subprocess.check_output(['git', *args], cwd=REPO)


def entries(ref):
    return {row.split(b'\t', 1)[1].decode(): row.split(b'\t', 1)[0].decode()
            for row in git('ls-tree', '-r', '-z', ref).split(b'\0') if row}


assert not git('status', '--porcelain').strip()
assert git('show', '-s', '--format=%P', SOURCE).decode().strip() == BASE
original = git('show', BASE + ':' + PATH).decode()
candidate = git('show', SOURCE + ':' + PATH).decode()
left, right = entries(BASE), entries(SOURCE)
changed = sorted(p for p in left.keys() | right.keys() if left.get(p) != right.get(p))
assert changed == [PATH]
assert left.keys() == right.keys()
assert all(left[p] == right[p] for p in left if p != PATH)
assert left[PATH].startswith('100644 blob ') and right[PATH].startswith('100644 blob ')
start = candidate.index('/** Reviews one funding message before confirming it through the existing finalized reconciliation path. */\n')
end = candidate.index('\nasync function nativeFlow() {', start)
helper = candidate[start:end]
assert helper.count('async function reviewedFund(') == 1 and helper.count('must(') == 13
reconstructed = candidate[:start] + candidate[end + 1:]
replacements = [
    ("import { isDeepStrictEqual } from 'node:util';\n", ''),
    ('const sendLog = [], signLog = [];', 'const sendLog = [];'),
    ('        signLog.push({ address: input.account.address, message: encodeBase64(message) });\n', ''),
    ("  const fund = await reviewedFund(client, trade, seller, 'native fund');",
     "  const fund = await act(client, trade, 'native fund', () => client.fund(trade, seller));"),
    ("  const fund = await reviewedFund(client, trade, session, 'spl fund', sellerAta.ata);",
     "  const fund = await act(client, trade, 'spl fund', () => client.fund(trade, session));"),
]
for before, after in replacements:
    assert reconstructed.count(before) == 1
    reconstructed = reconstructed.replace(before, after, 1)
assert reconstructed == original
with tempfile.TemporaryDirectory(prefix='g32-independent-native-syntax-') as temporary:
    path = Path(temporary) / 'client-run.mjs'
    path.write_text(candidate)
    result = subprocess.run(['node', '--check', str(path)], cwd=REPO, capture_output=True)
    assert result.returncode == 0 and result.stdout == b'' and result.stderr == b''
assert git('diff', '--check', '8e836baa877b676a863c877003e277b638fc5061..' + SOURCE) == b''
assert not git('status', '--porcelain').strip()
print(json.dumps({
    'source': SOURCE, 'sourceTree': git('rev-parse', SOURCE + '^{tree}').decode().strip(),
    'parent': BASE, 'sourcePath': PATH, 'changedPaths': changed,
    'sourceBytes': len(candidate.encode()), 'sourceSha256': hashlib.sha256(candidate.encode()).hexdigest(),
    'sourceGitEntry': right[PATH], 'unchangedOtherEntries': len(right) - 1,
    'nonDocumentationEntries': sum(not p.startswith('docs/') for p in right),
    'helperSha256': hashlib.sha256(helper.encode()).hexdigest(),
    'exactBaselineReconstruction': True, 'existingDriverRemainderByteIdentical': True,
    'syntaxCheckOnly': {'command': ['node', '--check', '<temporary exact Git source copy>'], 'exitCode': 0},
    'fullMainToSourceWhitespace': {'command': ['git', 'diff', '--check',
        '8e836baa877b676a863c877003e277b638fc5061..' + SOURCE], 'exitCode': 0},
    'nativeDriverEntryExecuted': False, 'validatorOrBrowserExecuted': False,
}, indent=2))
