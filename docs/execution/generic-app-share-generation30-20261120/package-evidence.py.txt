from pathlib import Path
import datetime
import gzip
import hashlib
import json
import re
import subprocess

ROOT = Path('/workspace/scratch/dcf03ef9bee8/p2p-generic-app-share-g30')
INPUT = Path('/workspace/scratch/dcf03ef9bee8/p2p-g30-evidence')
BUILD = Path('/workspace/scratch/dcf03ef9bee8/p2p-g30-build')
OUT = ROOT / 'docs/execution/generic-app-share-generation30-20261120'
BASE = 'fa7c1a11ff990be4646053731f1030d09eb8f9b6'
SOURCE = 'a2d69851430971ce51588c22c571e48b30b4ca01'
TREE = 'c484a9c5da4617667d9486d8c3bd4131db1ca8d6'
OUT.mkdir(parents=True, exist_ok=True)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)

def relative(path):
    return str(path.relative_to(ROOT))

assert git('rev-parse', 'HEAD').decode().strip() == SOURCE
assert git('rev-parse', 'HEAD^{tree}').decode().strip() == TREE
assert not git('diff', '--name-only', SOURCE)
artifacts = []

def retain(raw, name, kind, origin):
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(raw)
    row = {'path': relative(target), 'kind': kind, 'origin': origin, 'bytes': len(raw), 'sha256': sha(raw)}
    artifacts.append(row)
    return row

def compress(raw, name, kind, origin):
    packed = gzip.compress(raw, mtime=0)
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(packed)
    assert gzip.decompress(target.read_bytes()) == raw
    row = {'path': relative(target), 'kind': kind, 'origin': origin, 'rawBytes': len(raw), 'rawSha256': sha(raw),
           'gzipBytes': len(packed), 'gzipSha256': sha(packed), 'gzipMtime': 0}
    artifacts.append(row)
    return row

source_patch = git('diff', '--binary', BASE, SOURCE)
commands = []
for path in sorted(INPUT.glob('*.json')):
    raw = path.read_bytes()
    data = json.loads(raw)
    log = (INPUT / data['log']).read_bytes()
    patch = (INPUT / data['patch']).read_bytes()
    assert len(log) == data['logBytes'] and sha(log) == data['logSha256']
    assert sha(patch) == data['patchSha256']
    if data['label'] >= '04':
        for name, digest in data['files'].items():
            assert sha((ROOT / name).read_bytes()) == digest, (data['label'], name)
    if data['label'] in ['04-focused-composition', '05-ui-typecheck', '06-source-whitespace']:
        assert patch == source_patch
    if data['label'] >= '07':
        assert data['head'] == SOURCE and data['headTree'] == TREE and not patch
    counts = {key: int(value) for key, value in re.findall(r'^[ℹ#] (tests|suites|pass|fail|cancelled|skipped|todo) (\d+)$', log.decode(), re.M)}
    receipt = retain(raw, path.name, 'original-command-receipt', str(path))
    log_row = compress(log, data['log'] + '.gz', 'complete-command-output', str(INPUT / data['log']))
    patch_row = compress(patch, data['patch'] + '.gz', 'exact-pre-command-source-patch', str(INPUT / data['patch']))
    commands.append({'label': data['label'], 'argv': data['argv'], 'head': data['head'], 'headTree': data['headTree'],
                     'startedAt': data['startedAt'], 'endedAt': data['endedAt'], 'durationSeconds': data['durationSeconds'],
                     'exitCode': data['exitCode'], 'counts': counts, 'receipt': receipt['path'],
                     'output': log_row['path'], 'sourcePatch': patch_row['path']})

def entries(ref):
    result = {}
    for line in git('ls-tree', '-r', '-z', ref).split(b'\0'):
        if line:
            metadata, name = line.split(b'\t')
            result[name.decode()] = metadata.decode()
    return result
