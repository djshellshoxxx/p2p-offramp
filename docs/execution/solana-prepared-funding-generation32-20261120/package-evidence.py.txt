def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def entries(ref):
    result = {}
    for row in git('ls-tree', '-r', '-z', ref).split(b'\0'):
        if row:
            metadata, name = row.split(b'\t', 1)
            result[name.decode()] = metadata.decode()
    return result


assert git('rev-parse', 'HEAD').decode().strip() == SOURCE
assert git('rev-parse', 'HEAD^{tree}').decode().strip() == TREE
assert not git('diff', '--name-only', SOURCE)
assert git('diff', '--name-only', BASE, SOURCE).decode().splitlines() == PATHS
assert not OUT.exists(), 'Original evidence package must not be overwritten'
OUT.mkdir(parents=True)
artifacts = []


def retain(raw, name, kind, origin):
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(raw)
    row = {'path': str(target.relative_to(ROOT)), 'kind': kind, 'origin': origin,
           'bytes': len(raw), 'sha256': sha(raw)}
    artifacts.append(row)
    return row


def compress(raw, name, kind, origin):
    packed = gzip.compress(raw, mtime=0)
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(packed)
    assert gzip.decompress(target.read_bytes()) == raw
    assert int.from_bytes(packed[4:8], 'little') == 0
    row = {'path': str(target.relative_to(ROOT)), 'kind': kind, 'origin': origin,
           'rawBytes': len(raw), 'rawSha256': sha(raw), 'gzipBytes': len(packed),
           'gzipSha256': sha(packed), 'gzipMtime': 0}
    artifacts.append(row)
    return row

commands = []
original = {}
for path in sorted(INPUT.glob('[0-9][0-9]-*.json')):
    raw = path.read_bytes()
    data = json.loads(raw)
    original[data['label']] = data
    log = Path(data['log']['path']).read_bytes()
    patch = Path(data['patch']['path']).read_bytes()
    assert len(log) == data['log']['bytes'] and sha(log) == data['log']['sha256']
    assert len(patch) == data['patch']['bytes'] and sha(patch) == data['patch']['sha256']
    counts = {key: int(value) for key, value in re.findall(
        r'^[#ℹ]\s+(tests|pass|fail|cancelled|skipped|todo)\s+(\d+)', log.decode(), re.M)}
    assert counts == data['counts']
    if data['label'] in expected_counts:
        assert tuple(counts[key] for key in ['tests', 'pass', 'fail']) == expected_counts[data['label']]
        assert all(counts[key] == 0 for key in ['cancelled', 'skipped', 'todo'])
    assert data['exitCode'] == (1 if data['label'] in [
        '01-missing-capability-baseline', '05-frozen-solana-suite'] else 0)
    receipt = retain(raw, path.name, 'original-command-receipt', str(path))
    output = compress(log, Path(data['log']['path']).name + '.gz',
                      'complete-command-output', data['log']['path'])
    source_patch = compress(patch, Path(data['patch']['path']).name + '.gz',
                            'exact-pre-command-source-patch', data['patch']['path'])
    commands.append({'label': data['label'], 'command': data['command'], 'source': data['source'],
                     'tree': data['tree'], 'startedAt': data['startedAt'], 'completedAt': data['completedAt'],
                     'exitCode': data['exitCode'], 'counts': counts, 'receipt': receipt['path'],
                     'output': output['path'], 'sourcePatch': source_patch['path']})
