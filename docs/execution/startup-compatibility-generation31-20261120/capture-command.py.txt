import datetime
import hashlib
import json
import pathlib
import subprocess
import sys
import time

root = pathlib.Path('/workspace/scratch/dcf03ef9bee8/p2p-startup-compatibility-g31')
directory = pathlib.Path('/workspace/scratch/dcf03ef9bee8/p2p-g31-evidence')
directory.mkdir(exist_ok=True)
label, *command = sys.argv[1:]
if command and command[0] == '--':
    command.pop(0)
def git(*args):
    return subprocess.check_output(['git', *args], cwd=root)
patch = git('diff', '--binary', 'HEAD')
patch_path = directory / (label + '.patch')
assert not patch_path.exists(), 'Retained capture label must be unique'
patch_path.write_bytes(patch)
paths = [
    'apps/web/src/OfflineRoot.tsx', 'apps/web/src/ui/App.tsx',
    'apps/web/src/main.tsx', 'apps/web/src/ui/Pages.tsx',
    'apps/web/src/compatibility/model.ts', 'apps/web/src/compatibility/probe.ts',
    'apps/web/src/compatibility/startup.ts', 'apps/web/src/ui/StartupCompatibilityNotice.tsx',
    'apps/web/test/startup-compatibility-composition.test.mjs',
    'apps/web/test/generic-app-share-composition.test.mjs',
]
metadata = {
    'label': label, 'cwd': str(root), 'argv': command,
    'head': git('rev-parse', 'HEAD').decode().strip(),
    'headTree': git('rev-parse', 'HEAD^{tree}').decode().strip(),
    'patch': patch_path.name, 'patchSha256': hashlib.sha256(patch).hexdigest(),
    'files': {path: hashlib.sha256((root / path).read_bytes()).hexdigest() for path in paths if (root / path).exists()},
    'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
}
started = time.monotonic()
log_path = directory / (label + '.log')
with log_path.open('wb') as log:
    result = subprocess.run(command, cwd=root, stdout=log, stderr=subprocess.STDOUT)
metadata['durationSeconds'] = round(time.monotonic() - started, 3)
metadata['exitCode'] = result.returncode
metadata['endedAt'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
raw = log_path.read_bytes()
metadata['log'] = log_path.name
metadata['logBytes'] = len(raw)
metadata['logSha256'] = hashlib.sha256(raw).hexdigest()
(directory / (label + '.json')).write_text(json.dumps(metadata, indent=2) + '\n')
print(json.dumps({key: metadata[key] for key in ['label', 'exitCode', 'durationSeconds', 'log', 'logBytes', 'logSha256']}))
print(raw[-6500:].decode('utf-8', errors='replace'))
sys.exit(result.returncode)
