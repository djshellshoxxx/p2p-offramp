def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def entries(ref):
    result = {}

            metadata, path = line.split(b'\t', 1)
            result[path.decode()] = metadata.decode()
    return result

def capture(argv, stem):
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    result = subprocess.run(argv, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    completed = datetime.datetime.now(datetime.timezone.utc).isoformat()
    raw = result.stdout
    packed = gzip.compress(raw, mtime=0)
    (OUT / (stem + '.log.gz')).write_bytes(packed)
    data = {'argv': argv, 'source': SOURCE, 'sourceTree': TREE,
            'startedAt': started, 'completedAt': completed, 'exitCode': result.returncode,
            'file': stem + '.log.gz', 'rawBytes': len(raw), 'rawSha256': sha(raw),
            'gzipBytes': len(packed), 'gzipSha256': sha(packed), 'gzipMtime': 0}
    (OUT / (stem + '.json')).write_text(json.dumps(data, indent=2) + '\n')
    assert result.returncode == 0, raw.decode()
    return data
