def git(*args, env=None):
    return subprocess.check_output(['git', *args], cwd=REPO, env=env)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def entries(ref):
    return {p.decode(): meta.decode() for meta, p in
            (line.split(b'\t', 1) for line in git('ls-tree', '-r', '-z', ref).split(b'\0') if line)}


assert git('rev-parse', 'HEAD').decode().strip() == SOURCE
assert not git('status', '--porcelain').strip()
base, source = entries(BASE), entries(SOURCE)
changed = sorted(p for p in base.keys() | source.keys() if base.get(p) != source.get(p))
assert changed == ['packages/chain-solana/README.md', 'packages/chain-solana/src/funding-review.ts',
                   'packages/chain-solana/test/boundary.test.mjs', 'packages/chain-solana/test/funding-review.test.mjs']

assert git('diff', 'b8792ac4088f75603be8e0b40ba1506f2fb137f2', FORMATTED, '--numstat').decode().strip() == (
    '2\t1\tpackages/chain-solana/test/funding-review.test.mjs')

    '3\t1\tpackages/chain-solana/test/funding-review.test.mjs')

OUT.mkdir(parents=True)
artifacts = []


def save(name, raw, compress=False, provenance='original execution artifact'):
    data = gzip.compress(raw, mtime=0) if compress else raw
    path = OUT / name
    path.write_bytes(data)
    if compress:
        assert gzip.decompress(data) == raw and data[4:8] == b'\0' * 4
    artifacts.append({'path': str(path.relative_to(REPO)), 'bytes': len(data), 'sha256': digest(data),
                      'rawBytes': len(raw), 'rawSha256': digest(raw), 'gzipMtime': 0 if compress else None,
                      'provenance': provenance})


commands = []
snapshots = []
for receipt_path in sorted(LOGS.glob('[0-9][0-9]-*.json')):
    receipt = json.loads(receipt_path.read_text())
    label = receipt['label']
    raw_log = (LOGS / f'{label}.log').read_bytes()
    patch = (LOGS / f'{label}.patch').read_bytes()
    assert digest(raw_log) == receipt['log']['sha256'] and len(raw_log) == receipt['log']['bytes']
    assert digest(patch) == receipt['patch']['sha256'] and len(patch) == receipt['patch']['bytes']
    with tempfile.TemporaryDirectory(prefix='g33-evidence-index-') as directory:
        env = dict(os.environ, GIT_INDEX_FILE=str(Path(directory) / 'index'))
        subprocess.run(['git', 'read-tree', receipt['source']], cwd=REPO, env=env, check=True)
        if patch:
            subprocess.run(['git', 'apply', '--cached', '--binary', str(LOGS / f'{label}.patch')],
                           cwd=REPO, env=env, check=True)
        tree = git('write-tree', env=env).decode().strip()
    for name, expected in receipt['sourceFiles'].items():
        raw = git('show', f'{tree}:{name}')
        assert len(raw) == expected['bytes'] and digest(raw) == expected['sha256']
    snapshots.append({'label': label, 'head': receipt['source'], 'reconstructedTree': tree,
                      'verifiedSourceFiles': len(receipt['sourceFiles'])})
    save(f'{label}.json', receipt_path.read_bytes())
    save(f'{label}.log.gz', raw_log, True)
    save(f'{label}.patch.gz', patch, True)
    commands.append({k: receipt[k] for k in ['label', 'command', 'source', 'tree', 'startedAt', 'completedAt',
                                           'exitCode', 'counts']})
